<script>
    // 基础数据库
    const lastNames = ['苏', '叶', '林', '萧', '顾', '沈', '陆', '云', '柳', '白'];
    const firstNames = ['清寒', '红鱼', '浅雪', '琉璃', '如烟', '灵儿', '初见', '晚晴', '轻舞', '凝霜'];
    const fates = ['[绝色天仙]', '[天生剑胎]', '[桃花泛滥]', '[天煞孤星]', '[绝世炉鼎]', '[气运之子]', '[九阴绝脉]', '[红鸾星动]'];

    // 修为映射
    const cultivationLevels = [
        "凡人", "炼气·一层", "炼气·二层", "炼气·三层", "炼气·四层",
        "炼气·五层", "炼气·六层", "炼气·七层", "炼气·八层", "炼气·九层", "筑基·初期"
    ];

    // 年龄选择逻辑
    function selectAge(type, element) {
        document.querySelectorAll('.age-card').forEach(c => c.classList.remove('active'));
        element.classList.add('active');

        const slider = document.getElementById('cultivation-range');
        const desc = document.getElementById('past-story');
        const ageInput = document.getElementById('selected-age');

        if (type === 'child') {
            slider.max = 3;
            slider.value = 1;
            desc.placeholder = "虽带宿慧，然身躯稚嫩，前世记忆如同迷梦...";
            ageInput.value = "幼年期 (4-6岁)";
        } else if (type === 'teen') {
            slider.max = 8;
            slider.value = 4;
            desc.placeholder = "豆蔻年华，初识仙途，前世的执念让你显得早熟...";
            ageInput.value = "少年期 (7-12岁)";
        } else {
            slider.max = 10;
            slider.value = 6;
            desc.placeholder = "你是何人？因何而终？保留了何种执念？...";
            ageInput.value = "青年期 (13-18岁)";
        }
        updateCultivation(slider.value);
    }

    // 随机名字
    function randomName() {
        const ln = lastNames[Math.floor(Math.random() * lastNames.length)];
        const fn = firstNames[Math.floor(Math.random() * firstNames.length)];
        document.getElementById('current-name').value = ln + fn;
    }

    // 随机命格
    function randomFate() {
        // 随机取2个命格
        const shuffled = fates.sort(() => 0.5 - Math.random());
        document.getElementById('fate').value = shuffled.slice(0, 2).join(' ');
    }

    // 更新修为显示
    function updateCultivation(val) {
        const text = cultivationLevels[val] || "未知";
        document.getElementById('cultivation-val').innerText = text;
    }

    // 提交逻辑
    function submitCreation() {
        const data = {
            pastName: document.getElementById('past-name').value || "未知",
            pastStory: document.getElementById('past-story').value || "无",
            age: document.getElementById('selected-age').value,
            curName: document.getElementById('current-name').value || "汖星",
            sect: document.getElementById('sect').value,
            body: document.getElementById('constitution').value,
            art: document.getElementById('cultivation-method').value,
            cultivation: document.getElementById('cultivation-val').innerText,
            fate: document.getElementById('fate').value
        };

        // 构建发送给AI的文本
        const message = `[系统指令：用户角色已创建]
姓名：${data.curName}
骨龄：${data.age}
宗门：${data.sect}
修为：${data.cultivation}
体质：${data.body}
功法：${data.art}
命格：${data.fate}
---
前世名讳：${data.pastName}
前世经历：${data.pastStory}`;

        // 使用SillyTavern的命令发送
        // 1. 发送设定文本给AI (不触发回复，或者是让AI根据这个设定开始剧情)
        // 这里选择直接触发AI回复，让AI根据设定开始第一段剧情
        const command = `/send ${message} | /trigger`;

        if (window.triggerSlash) {
            window.triggerSlash(command);
        } else {
            alert("请在SillyTavern中运行此界面。\n\n模拟发送内容：\n" + message);
        }
    }
</script>
