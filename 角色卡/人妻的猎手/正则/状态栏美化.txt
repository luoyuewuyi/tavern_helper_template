  <script type="module">
    const HEROINES = [
      { key: '陆沉雪', label: '沉雪' },
      { key: '陆美玲', label: '美玲' },
      { key: '柳如玉', label: '如玉' },
      { key: '何美兰', label: '美兰' },
      { key: '李淑慧', label: '淑慧' },
      { key: '张雪', label: '张雪' },
      { key: '白曼文', label: '曼文' }
    ];

    function populateData() {
      const all = getAllVariables();
      const sd = _.get(all, 'stat_data', {});

      $('#world-time').text(_.get(sd, '世界.当前时间', 'N/A'));
      $('#world-date').text(_.get(sd, '世界.当前日期', 'N/A'));

      HEROINES.forEach(h => {
        const d = _.get(sd, h.key, {});
        const val = _.get(d, '❤️攻略值', 0);
        const stage = _.get(d, '攻略阶段', 'N/A');
        const body = _.get(d, '身体状态', {});

        $(`#${h.key}-val`).text(val);
        $(`#${h.key}-fill`).css('width', val + '%');
        $(`#${h.key}-stage`).text(stage);
        $(`#${h.key}-clothes`).text(_.get(body, '👚衣着', 'N/A'));
        $(`#${h.key}-mouth`).text(_.get(body, '👄口腔', 'N/A'));
        $(`#${h.key}-breast`).text(_.get(body, '🍈胸乳', 'N/A'));
        $(`#${h.key}-pussy`).text(_.get(body, '🌸小穴', 'N/A'));
        $(`#${h.key}-anal`).text(_.get(body, '💮菊穴', 'N/A'));
        $(`#${h.key}-womb`).text(_.get(body, '🏠子宫', 'N/A'));
        $(`#${h.key}-pose`).text(_.get(d, '💃体态姿势', 'N/A'));
        $(`#${h.key}-mind`).text(_.get(d, '💭内心想法', 'N/A'));
      });
    }

    function buildTabs() {
      const $tabs = $('#tabs');
      const $contents = $('#contents');

      HEROINES.forEach((h, i) => {
        const $btn = $('<div>')
          .addClass('tab-btn')
          .addClass(i === 0 ? 'active' : '')
          .text(h.label)
          .attr('data-key', h.key);
        $tabs.append($btn);

        const html = `
          <div class="tab-content ${i === 0 ? 'active' : ''}" data-key="${h.key}">
            <div class="card">
              <div class="card-header">
                <span class="card-name">${h.key}</span>
                <span class="card-stage" id="${h.key}-stage"></span>
              </div>
              <div class="progress-label">❤️ 攻略值 <span id="${h.key}-val">0</span>/100</div>
              <div class="progress-wrap"><div class="progress-fill" id="${h.key}-fill"></div></div>
              <div class="divider"></div>
              <div class="field-row"><div class="field-label">👚 衣着</div><div class="field-value" id="${h.key}-clothes"></div></div>
              <div class="field-row"><div class="field-label">👄 口腔</div><div class="field-value" id="${h.key}-mouth"></div></div>
              <div class="field-row"><div class="field-label">🍈 胸乳</div><div class="field-value" id="${h.key}-breast"></div></div>
              <div class="field-row"><div class="field-label">🌸 小穴</div><div class="field-value" id="${h.key}-pussy"></div></div>
              <div class="field-row"><div class="field-label">💮 菊穴</div><div class="field-value" id="${h.key}-anal"></div></div>
              <div class="field-row"><div class="field-label">🏠 子宫</div><div class="field-value" id="${h.key}-womb"></div></div>
              <div class="divider"></div>
              <div class="field-row"><div class="field-label">💃 体态姿势</div><div class="field-value" id="${h.key}-pose"></div></div>
              <div class="field-row"><div class="field-label">💭 内心想法</div><div class="field-value" id="${h.key}-mind"></div></div>
            </div>
          </div>
        `;
        $contents.append(html);
      });

      $tabs.on('click', '.tab-btn', function () {
        const key = $(this).attr('data-key');
        $('.tab-btn').removeClass('active');
        $(this).addClass('active');
        $('.tab-content').removeClass('active');
        $(`.tab-content[data-key="${key}"]`).addClass('active');
      });
    }

    async function init() {
      await waitGlobalInitialized('Mvu');
      buildTabs();
      populateData();

      eventOn(Mvu.events.VARIABLE_UPDATE_ENDED, () => {
        populateData();
      });
    }

    $(errorCatched(init));
  </script>
