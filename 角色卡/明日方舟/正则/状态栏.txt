<script>
    function sendOption(element) {
        // 获取选项内容
        const content = element.getAttribute('data-content');
        if (!content) return;

        // 构建指令：/send [内容] | /trigger
        // 这里的 /trigger 会让AI立即回复
        const command = `/send ${content} | /trigger`;

        // 调用酒馆助手的执行函数

            triggerSlash(command);
        } else {

    }
