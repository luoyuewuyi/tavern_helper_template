    // 转义
    function escapeSpecialChars(str) {
        if (!str) return "无";
        return str.replace(/\|/g, '\\|').replace(/"/g, '\\"').replace(/\n/g, '\\n');
    }

    // 提交数据
    function submitTerminalData() {
        // 1. 获取数据
        const name = document.getElementById('user-name').value.trim();
        const race = document.getElementById('user-race').value.trim();
        const identity = document.getElementById('user-identity').value.trim(); // 现在是文本框了
        const gender = document.querySelector('input[name="gender"]:checked').value;
        const appearance = document.getElementById('user-appearance').value.trim();
        const background = document.getElementById('user-background').value.trim();
        const scenario = document.getElementById('scenario-input').value.trim();

        // 2. 校验
        if (!name) { alert("【系统错误】\n检测到代号为空，无法建立档案索引。"); return; }
        if (!scenario) { alert("【系统错误】\n未检测到神经连接情境，请选择时间锚点或手动输入。"); return; }

        // 3. 构建完整的 Prompt (不再写入世界书，而是直接拼接)
        // 格式：[角色档案] + [详细设定] + [开局剧情]
        let fullPrompt = `[系统指令：立即载入以下用户角色档案与当前情境，开始剧情模拟]\n\n`;

        fullPrompt += `【用户档案】\n`;
        fullPrompt += `代号：${name}\n`;
        fullPrompt += `性别：${gender}\n`;
        fullPrompt += `种族：${race || "未知"}\n`;
        fullPrompt += `身份：${identity || "未知"}\n`;

        if (appearance) fullPrompt += `外貌：${appearance}\n`;
        if (background) fullPrompt += `背景/技能：${background}\n`;

        fullPrompt += `\n【当前情境】\n${scenario}`;

        // 4. 发送指令
        // 使用 /send 发送内容，使用 /trigger 触发 AI 回复
        // 注意：这里没有 /setentryfield 了

        // 为了防止 prompt 太长导致 URL 编码问题或 ST 解析问题，我们做简单的转义
        // 但通常 ST 的 /send 可以处理长文本

        // 界面反馈
        const command = `/echo severity=success 正在同步神经连接... | /send ${escapeSpecialChars(fullPrompt)} | /trigger`;

        if (typeof triggerSlash === 'function') {
            triggerSlash(command);

            // 可选：隐藏界面或显示加载遮罩
            // document.querySelector('.terminal-body').style.opacity = '0.5';
        } else {
            alert("错误：酒馆助手(STHelper)未响应。");
            console.log(command);
        }
    }
