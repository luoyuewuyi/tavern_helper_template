        /* 核心驱动与渲染 */
        async function init() {
            if (typeof waitGlobalInitialized === 'function') { await waitGlobalInitialized('Mvu'); }
            renderAll();
            // 订阅 MVU 变量更新事件，替代固定轮询
            if (typeof Mvu !== 'undefined' && Mvu.events && Mvu.events.VARIABLE_UPDATE_ENDED) {
                Mvu.events.VARIABLE_UPDATE_ENDED.on(renderAll);
            }
            // 兜底轮询
            setInterval(renderAll, 3000);
        }

        function getStatData() {
            // 从最新楼层反向查找有 MVU 数据的楼层
            try {
                if (typeof Mvu !== 'undefined' && Mvu.getMvuData) {
                    for (let i = -1; i >= -20; i--) {
                        const mvuData = Mvu.getMvuData({ type: 'message', message_id: i });
                        const d = _.get(mvuData, 'stat_data');
                        if (d && !_.isEmpty(d)) return d;
                    }
                }
            } catch(e) {}
            // 兜底：当前楼层变量
            const allVars = typeof getAllVariables === 'function' ? getAllVariables() : {};
            return _.get(allVars, 'stat_data', {});
        }

        /* 时间格式化转换 (-) -> (:) */
        function formatClock(str) {
            return str ? str.replace(/-/g, ':') : '--';
        }

        function renderAll() {
            const d = getStatData();

            /* 1. 顶部基础信息面板 */
            $('#main-time').text(formatClock(_.get(d, '时间系统.主世界时间')));
            $('#p-time').text(formatClock(_.get(d, '时间系统.个人时间')));
            $('#weather').text(_.get(d, '世界信息.当前天气', '晴'));

            /* 2. 肖像经验环驱动 */
            const lvl = _.get(d, '角色概览.等级', 1);
            const xp = _.get(d, '角色概览.经验值', 0);
            const maxXp = _.get(d, '角色概览.最大经验值', lvl * 100);
            const xpPct = Math.min(100, (xp / maxXp) * 100);
            const offset = 440 - (440 * xpPct / 100);
            $('.exp-bar').css('stroke-dashoffset', offset);
            $('#val-lvl').text(lvl);

            /* 3. 五维核心数值卡 */
            $('#val-hp').text(`${_.get(d, '角色概览.生命值', 100)} / 100`);
            $('#val-mp').text(`${_.get(d, '角色概览.法力值', 100)} / 100`);
            $('#val-age').text(_.get(d, '角色概览.年龄', 18));
            $('#val-life').text(_.get(d, '角色概览.寿命', 29200));

            const dMax = _.get(d, '角色概览.最大欲望值', dLvl * 500);
            const dPct = Math.floor((dVal / dMax) * 100);
            $('.desire-fill').css('width', dPct + '%');
            $('#desire-lvl-txt').text(`欲望等级 LV.${dLvl}`);
            $('#desire-pct-txt').text(dPct + '%');

            renderSocial(d);
            renderCombat(d);
            renderGear(d);
            renderInventory(d);
            renderPrivateInfo(d);

            $('#mod-list').html(renderMap(m, (v, k) => `<div class="list-item" style="margin:0" onmouseover="tip(event, '<b>${k}</b><br>${v.描述}<br>来源:${v.来源}')" onmouseout="untip()">${k}</div>`));
        }

        /* 辅助工具：槽位与品质 */
        function renderSlot(slot, item, isRelic=false) {
            if(!item) return `<div class="list-item" style="opacity:0.3"><span>${slot}</span><span>[ Empty ]</span></div>`;
            const qMap = { '普通': 'white', '稀有': 'green', '史诗': 'purple', '传奇': 'yellow', '神器': 'red' };
            const qClass = isRelic ? 'q-relic' : `q-${qMap[item.品质] || 'white'}`;
            const tipTxt = isRelic ? 
                `<b>${item.名称}</b> (Lv.${item.圣器等级})<br>${item.描述}<br>存储:${item.存储情况 || item.当前存储量}/${item.最大存储量}<br>功能:${item.功能}` :
                `<b>${item.名称}</b><br>${item.简介}<br>效果:${item.效果}<br>外观:${item.外观}`;
            
            return `<div class="list-item" onmouseover="tip(event, '${tipTxt}')" onmouseout="untip()"><span>${slot}</span><span class="${qClass}">${item.名称}</span></div>`;
        }

        function renderMap(obj, fn) {
            if(_.isEmpty(obj)) return '<div style="font-size:0.75rem; color:var(--text-dim)">[ 无活动数据 ]</div>';
            return _.map(obj, fn).join('');
        }

        /* 鼠标悬浮 */
        window.tip = (e, txt) => {
            const $t = $('#tooltip');
            $t.html(txt).css({ top: e.clientY + 15, left: e.clientX + 15 }).show();
        };
        window.untip = () => $('#tooltip').hide();

        /* 更换肖像 */
        window.changePortrait = () => {
            const defaultImg = "https://files.catbox.moe/cyhi5w.png"
            const url = prompt("请输入肖像图片链接 (URL):");
            if (url) $('.portrait-img').attr('src', url);
        };

        $(() => {
            init();
            $('.tab-btn').on('click', function() {
                const target = $(this).data('target');
                $('.tab-btn, .pane').removeClass('active');
                $(this).addClass('active');
                $(`#pane-${target}`).addClass('active');
            });
        });
    </script>
