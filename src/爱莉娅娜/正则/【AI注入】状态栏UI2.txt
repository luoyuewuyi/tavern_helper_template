        /* 辅助工具：装备槽渲染 */
        function renderSlot(name, item, isRelic=false) {
            if(!item) return `<div class="list-item" style="opacity:0.3"><span>${name}</span><span>[ Empty ]</span></div>`;
            const q = isRelic ? 'relic' : getQNum(item.品质);
            const tipTxt = isRelic ? 
                `<b>${item.名称}</b> (Lv.${item.圣器等级})<br>${item.描述}<br>功能:${item.功能}<br>内部存储:${item.当前存储量}/${item.最大存储量}` :
                `<b>${item.名称}</b> [${item.品质}]<br>${item.简介}<br>效果:${item.效果}<br>外观:${item.外观}`;
            return `<div class="list-item" onmouseover="tip(event, '${tipTxt}')" onmouseout="untip()"><span>${name}</span><span class="q-${q}">${item.名称}</span></div>`;
        }

        function getQNum(quality) {
            const m = {'普通':1,'稀有':2,'史诗':3,'传奇':4,'神器':5};
            return m[quality] || 1;
        }

        function renderMap(obj, fn) {
            if(_.isEmpty(obj)) return '<div style="font-size:0.7rem; color:var(--text-dim)">无数据</div>';
            return _.map(obj, fn).join('');
        }

        window.untip = () => $('#tooltip').hide();

            });
        });
