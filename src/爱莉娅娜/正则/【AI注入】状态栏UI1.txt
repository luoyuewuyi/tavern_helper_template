        function renderGearPane(data) {

        function renderSlot(slotName, item, isRelic=false) {
            if(!item) return `<div class="list-item" style="opacity:0.3"><span>${slotName}</span><span>[ 空 ]</span></div>`;
            const qClass = isRelic ? 'q-relic' : getQClass(item.品质);
            const detail = isRelic ? 
                `圣器级:${item.圣器等级}<br>${item.描述}<br>存储:${item.存储情况 || item.当前存储量}/${item.最大存储量}<br>功能:${item.功能}` :
                `品质:${item.品质}<br>${item.简介}<br>效果:${item.效果}<br>外观:${item.外观}`;
            return `<div class="list-item" onmouseover="showTip(event, '${detail}')" onmouseout="hideTip()"><span>${slotName}</span><span class="${qClass}">${item.名称}</span></div>`;
        }
