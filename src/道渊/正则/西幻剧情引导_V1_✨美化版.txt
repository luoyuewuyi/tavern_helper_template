    function extractTagFromChat(tagName){
        try{
            var chat = null;
            if(typeof getContext === 'function'){ try{ var ctx = getContext(); chat = ctx && ctx.chat ? ctx.chat : null; }catch(e){} }
            if(!chat && typeof window.SillyTavern !== 'undefined') chat = window.SillyTavern.chat ? window.SillyTavern.chat : null;
            if(!chat) return '';
            var regex = new RegExp('<'+tagName+'>([\\s\\S]*?)<\\/'+tagName+'>', 'i');
            for(var i = chat.length - 1; i >= 0; i--){
                var msg = chat[i]; if(!msg) continue;
                var sources = [msg.extra && msg.extra.qrf_plot ? msg.extra.qrf_plot : null, msg.mes, msg.qrf_plot];
                if(msg.swipes && Array.isArray(msg.swipes)){ for(var s = 0; s < msg.swipes.length; s++){ var swipe = msg.swipes[s]; if(typeof swipe === 'string') sources.push(swipe); else if(swipe && swipe.extra && swipe.extra.qrf_plot) sources.push(swipe.extra.qrf_plot); } }

            }
        }catch(e){}
        return '';

    function parseLegitimacy(raw) {
        var r = { result: '', items: [], summary: '' };
        if(!raw.trim()) return r;
        var resultM = raw.match(/##\s*审查结果[：:]\s*(✅\s*通过|❌\s*不通过|通过|不通过)/i);

        var m;
        while((m = itemPattern.exec(raw)) !== null) {
            r.items.push({ title: m[1].trim(), status: m[2].trim(), desc: m[3].trim() });
        }
        var summaryM = raw.match(/##\s*综合评价\s*\n([\s\S]*?)(?=<\/legitimacy_check>|$)/i);
        if(summaryM) r.summary = summaryM[1].trim();

    function parseCharacterAnalysis(raw) {
        var r = { chars: [], note: '' };

        var charPattern = /###\s*📊\s*([^\n]+)\n([\s\S]*?)(?=###\s*📊|---+|✅\s*分析完成|$)/gi;
        var m;
        while((m = charPattern.exec(raw)) !== null) {
            var name = m[1].trim();
            var content = m[2].trim();
            if(name && content) {
                var charData = { name: name, raw: content };
                var infoPatterns = [
                    { key: '基础信息', pattern: /\*\*基础信息\*\*[：:]?\s*([^\n]+)/i },
                    { key: '当前状态', pattern: /\*\*当前状态\*\*[：:]?\s*([^\n]+)/i },
                    { key: '心理状态', pattern: /\*\*心理状态\*\*[：:]?\s*([^\n]+)/i },
                    { key: '认知边界', pattern: /\*\*认知边界\*\*[：:]?\s*([^\n]+)/i },
                    { key: '行为预测', pattern: /\*\*行为预测\*\*[：:]?\s*([^\n]+)/i },
                    { key: '情绪优化', pattern: /\*\*情绪优化\*\*[：:]?\s*([^\n]+)/i },
                    { key: '主动性建议', pattern: /\*\*主动性建议\*\*[：:]?\s*([^\n]+)/i }
                ];
                charData.info = {};
                infoPatterns.forEach(function(p) {
                    var match = content.match(p.pattern);
                    if(match) charData.info[p.key] = match[1].trim();
                });
                r.chars.push(charData);
            }
        }
        var noteM = raw.match(/\*\*核心原则\*\*[：:]?\s*([^\n]+)/i);
        if(noteM) r.note = noteM[1].trim();
        return r;

    /* ========== Reason Parser ========== */
    function parseReason(raw) {
        var r = { items: [], total: 0 };
        if(!raw.trim()) return r;
        var itemPattern = /-\s*(AM\d+)\s*[：:]\s*([^\n]+)/gi;
        var m;
        while((m = itemPattern.exec(raw)) !== null) {
            r.items.push({ code: m[1].trim(), desc: m[2].trim() });
        }
        var totalM = raw.match(/✅\s*合计[：:]\s*(\d+)\s*条/);
        if(totalM) r.total = parseInt(totalM[1]);
        return r;
    }

    function initReason() {
        var container = document.getElementById('mem-content');
        var raw = extractReasonFromChat();
        if(!raw.trim()) return;
        var parsed = parseReason(raw);
        if(parsed.items.length === 0) return;
        var html = '<div class="reason-section">';
        html += '<div class="reason-title">📋 召回理由</div>';
        html += '<div class="reason-list">';
        parsed.items.forEach(function(item) {
            html += '<div class="reason-item"><span class="reason-code">' + escapeHtml(item.code) + '</span><span class="reason-desc">' + escapeHtml(item.desc) + '</span></div>';
        });
        html += '</div>';
        if(parsed.total) html += '<div style="text-align:center;color:var(--tn-comment);font-size:11px;margin-top:8px;">✅ 合计：' + parsed.total + '条理由</div>';
        html += '</div>';
        container.innerHTML += html;
    }

    /* ========== Supplement Parser ========== */
    function parseSupplement(raw) {
        var items = [];
        if(!raw.trim()) return items;
        var lines = raw.split('\n');
        lines.forEach(function(l) {
            var m = l.match(/-\s*\[([^\]]+)\]\s*(.+)/);
            if(m) items.push({ type: m[1].trim(), content: m[2].trim() });
        });
        return items;
    }

    function initSupplement() {
        var container = document.getElementById('supplementContent');
        var parsed = parseSupplement(extractSupplementFromChat());
        if(parsed.length === 0) { container.innerHTML = '<div class="empty-state">暂无补充信息</div>'; return; }
        var html = '<div class="supplement-section">';
        html += '<div class="supplement-title">✨ 补充信息 (' + parsed.length + '条)</div>';
        parsed.forEach(function(item) {
            html += '<div class="supplement-item">';
            html += '<div class="supplement-type">[' + escapeHtml(item.type) + ']</div>';
            html += '<div class="supplement-content">' + escapeHtml(item.content) + '</div>';
            html += '</div>';
        });
        html += '</div>';
        container.innerHTML = html;
    }
